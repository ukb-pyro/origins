  }

    this.context.push(data);
    this.pruneContext();
  }

    while (this.context.length > this.bandwidthLimit) {
      this.context.shift(); // Drop oldest info - pruning
    }
  }

    // Bayesian update: simple weighted average for illustration

    this.act();
